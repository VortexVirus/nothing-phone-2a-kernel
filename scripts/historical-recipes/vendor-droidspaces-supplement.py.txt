#!/usr/bin/env python3
"""DroidSpaces companion modules; official Pacman flags, combined GKI, no flash."""
from pathlib import Path
import json,sys,subprocess,os,hashlib,datetime,shutil,re
p=Path('${PROJECT_ROOT}');base,run=sys.argv[1:3]
assert all(re.fullmatch('[a-z0-9-]+',x) for x in [base,run])
i=json.loads((p/'artifacts/vendor-droidspaces'/base/'BUILD-INFO.json').read_text())
assert i['exit_code']==0
for name,digest in i['gki_sha256'].items():
 assert hashlib.sha256((Path(i['gki_input'])/name).read_bytes()).hexdigest()==digest
assert hashlib.sha256((Path(i['out_dir'])/'kernel-5.15/.config').read_bytes()).hexdigest()==i['config_sha256']
out=Path(i['out_dir']);k=out/'kernel-5.15'
for other in (p/'artifacts/vendor-droidspaces').glob('*/BUILD-INFO.json'):
 v=json.loads(other.read_text());assert not(v.get('out_dir')==str(out) and v.get('exit_code') is None)
for name,sha in i['refs'].items():
 assert subprocess.check_output(['git','-C',str(p/'sources'/name),'rev-parse','HEAD'],text=True).strip()==sha
 assert not subprocess.check_output(['git','-C',str(p/'sources'/name),'status','--porcelain','--untracked-files=no'],text=True).strip()
a=p/'artifacts/vendor-droidspaces'/run;a.mkdir(exist_ok=False)
info=dict(i,date_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),base_run=base,stage='supplementary DroidSpaces modules against combined GKI; not phone validated',exit_code=None,commands=[],phone_modified=False)
meta=a/'BUILD-INFO.json'
def save():meta.write_text(json.dumps(info,indent=2)+'\n')
save()
env=os.environ.copy();env['PATH']=str(p/'toolchains/clang-r450784e/bin')+':'+str(p/'sources/gki/prebuilts/kernel-build-tools/linux-x86/bin')+':/usr/bin:/bin'
common=['make','-C',str(p/'sources/nothing-vendor-stock'),'O='+str(k),'LLVM=1','LLVM_IAS=1','ARCH=arm64','TOP='+str(p/'sources/vendor-droidspaces-platform'),'KBUILD_MIXED_TREE='+i['gki_input'],'-j4','TARGET_BUILD_VARIANT=user']
wifi='connectivity/wlan/core/gen4m';msync='msync2_frd_cus';bt='connectivity/bt/mt66xx/btif'
# Module symbols required by the official Android.mk LOCAL_REQUIRED_MODULES.
symvers=[out/'nothing-modules-stock'/x/'Module.symvers' for x in ['connectivity/conninfra','connectivity/connfem','connectivity/wlan/adaptor/build/connac2x']]
assert all(x.is_file() for x in symvers), symvers
wifi_opts=['CONFIG_MTK_COMBO_WIFI_HIF=axi','MODULE_NAME=wlan_drv_gen4m_6886','MTK_COMBO_CHIP=CONNAC2X2_SOC7_0','WLAN_CHIP_ID=6886','MTK_ANDROID_WMT=y','WIFI_IP_SET=1','MTK_ANDROID_EMI=y','CONNAC_VER=2_0','MTK_WLAN_SERVICE=yes','KBUILD_EXTRA_SYMBOLS='+' '.join(map(str,symvers))]
# Keep the source blob intact; place its exact bytes in the object directory.
blob=p/'sources/nothing-modules-stock/msync2_frd_cus/src/mtk_disp_msync.o'
blobout=out/'nothing-modules-stock/msync2_frd_cus/src/mtk_disp_msync.o'
blobout.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(blob,blobout)
info['prebuilt_msync_sha256']=hashlib.sha256(blob.read_bytes()).hexdigest();save()
# These Android Makefiles assume an absolute M/src directory. Keep a small
# local build-input snapshot, not a second repository or writes into the source worktree.
info['local_source_snapshots']={}
for module in [wifi,msync,'connectivity/bt/mt66xx']:
 source=p/'sources/nothing-modules-stock'/module;target=out/'nothing-modules-stock'/module
 manifest={str(f.relative_to(source)):hashlib.sha256(f.read_bytes()).hexdigest() for f in sorted(source.rglob('*')) if f.is_file()}
 shutil.copytree(source,target,dirs_exist_ok=True)
 assert all(hashlib.sha256((target/name).read_bytes()).hexdigest()==sha for name,sha in manifest.items())
 info['local_source_snapshots'][module]=manifest
save()
bt_opts=['BT_PLATFORM=6886','TARGET_BUILD_VARIANT=user','KBUILD_EXTRA_SYMBOLS='+' '.join(map(str,symvers[:2]))]
with (p/'logs/vendor-droidspaces'/('build-'+run+'.log')).open('w') as log:
 for module,opts in [('connectivity/conninfra',[]),(wifi,wifi_opts),(msync,[]),(bt,bt_opts)]:
  (out/'nothing-modules-stock'/module).mkdir(parents=True,exist_ok=True)
  module_path='../nothing-modules-stock/'+module if module=='connectivity/conninfra' else str(out/'nothing-modules-stock'/module)
  cmd=common+['M='+module_path]+opts+['modules']
  info['commands'].append(cmd);save();log.write(json.dumps(cmd)+'\n');log.flush()
  r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
  if r.returncode:break
info['exit_code']=r.returncode;info['config_sha256_after']=hashlib.sha256((k/'.config').read_bytes()).hexdigest();save()
print(json.dumps(info,indent=2));sys.exit(r.returncode)
