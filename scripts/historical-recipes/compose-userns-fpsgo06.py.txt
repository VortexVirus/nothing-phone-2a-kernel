for row in m['modules']:
 if row['name']=='mtk_fpsgo':row.update(check['modules']['candidate'])

(O/'DO-NOT-FLASH.txt').write_text('Modules only. Packaging and physical validation required. No automatic flash.\n')

 (a/'SHA256SUMS').write_text(''.join(sha(f)+'  '+str(f.relative_to(a))+'\n' for f in sorted(a.rglob('*')) if f.is_file() and f.name!='SHA256SUMS'))
print(O)
