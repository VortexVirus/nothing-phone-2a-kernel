#!/usr/bin/env python3
"""Offline ELF symbol availability; neither load-order nor runtime validation."""
from pathlib import Path
import struct, json
p=Path('${PROJECT_ROOT}')
mods=list((p/'stock-images/vendor-modules').glob('*.ko'))+list((p/'stock-images/vendor-dlkm').rglob('*.ko'))
rows=[]; providers={}
for f in mods:
 b=f.read_bytes()
 assert b[:6]==b'\x7fELF\x02\x01'
 off=struct.unpack_from('<Q',b,40)[0]
 ents,n,idx=struct.unpack_from('<HHH',b,58)
 hs=[struct.unpack_from('<IIQQQQIIQQ',b,off+i*ents) for i in range(n)]
 imports=[]; exports=[]
 for h in hs:
  if h[1]!=2: continue
  sh=hs[h[6]]; strings=b[sh[4]:sh[4]+sh[5]]
  for j in range(h[4],h[4]+h[5],h[9]):
   name,info,other,section,val,size=struct.unpack_from('<IBBHQQ',b,j)
   name=strings[name:].split(b'\0')[0].decode()
   if name.startswith('__ksymtab_'): exports.append(name[len('__ksymtab_'):])
   if section==0 and name: imports.append({'name':name,'weak':info>>4==2})
 for name in exports: providers.setdefault(name,[]).append(str(f.relative_to(p)))
 rows.append({'path':str(f.relative_to(p)),'imports':imports,'signed':b.endswith(b'~Module signature appended~\n')})
gki={l.split()[1] for l in (p/'artifacts/stock/pipeline-01/vmlinux.symvers').read_text().splitlines()}
missing=[{'module':r['path'],**i} for r in rows for i in r['imports'] if i['name'] not in gki and i['name'] not in providers]
report={'scope':__doc__,'modules':len(mods),'signed_modules':sum(r['signed'] for r in rows),'vendor_export_names':len(providers),'missing_imports':missing,'duplicate_vendor_exports':{k:v for k,v in providers.items() if len(v)>1},'gki_protected_symbol_list_empty':len((p/'sources/common/android/abi_gki_modules_protected').read_text().splitlines())==1,'gki_protected_export_list_empty':len((p/'sources/common/android/abi_gki_modules_exports').read_text().splitlines())==1,'system_dlkm_modules':len(list((p/'stock-images/system-dlkm').rglob('*.ko')))}
(p/'logs/repro/vendor-symbol-availability.json').write_text(json.dumps(report,indent=2)+'\n')
print('Modules',len(mods),'missing imports',len(missing))
