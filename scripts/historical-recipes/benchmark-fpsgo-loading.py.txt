#!/usr/bin/env python3
"""Host-only microbenchmark of extracted FPSGO loading snapshots, not Android/SLUB."""
from pathlib import Path
import subprocess,os,json,sys,re,hashlib,platform,statistics,datetime
P=Path('${PROJECT_ROOT}');run=sys.argv[1];assert re.fullmatch('[a-z0-9-]+',run)
L=P/'logs/performance-fpsgo'/('benchmark-'+run);L.mkdir(exist_ok=False)
sourcefile=P/'logs/performance-fpsgo/loading-tests-grouped-07/loading.c'
s=sourcefile.read_text().split('struct result{')[0]
a=s.index('static int allocs,');b=s.index('#define spin_lock_irqsave',a)
s=s[:a]+'''static void *alloc(size_t n,size_t z){return calloc(n,z);}
#define kcalloc(n,z,g) alloc(n,z)
#define kzalloc(n,g) alloc(1,n)
static void kfree(void*p){free(p);}
'''+s[b:]
a=s.index('#define spin_lock_irqsave');b=s.index('/* Per-call heap snapshots;',a)
s=s[:a]+r'''
/* Compiler barriers retain snapshot ordering, but DO NOT emulate kernel locks. */
#define spin_lock_irqsave(p,f) do {(f)=0;__asm__ volatile("" ::: "memory");}while(0)
#define spin_unlock_irqrestore(p,f) do {(void)(f);__asm__ volatile("" ::: "memory");}while(0)
#define xgf_trace(...) ((void)0)
#define fpsgo_main_trace(...) ((void)0)

'''+s[b:]
for name in ['old_global','new_global','old_cluster','new_cluster']:s=s.replace('static int '+name+'(', 'static __attribute__((noinline)) int '+name+'(')
s='#define _GNU_SOURCE\n#include <time.h>\n'+s
s+=r'''
static volatile unsigned long long sink;
static unsigned long long ns(void){struct timespec t;assert(!clock_gettime(CLOCK_MONOTONIC_RAW,&t));return (unsigned long long)t.tv_sec*1000000000ULL+t.tv_nsec;}
static void setup(int scenario){
 cluster_num=2;test_mode=1;last_cb_ts=2560;clus_obv[0]=73;clus_obv[1]=85;
 for(int i=0;i<LOADING_CNT;i++){
  prev_cb_ts[i]=i*10;lastest_ts[i]=(i+1)*10;obvs[i]=73;
  lastest_obv_cl[i]=obs_store[i];lastest_is_cl_isolated[i]=is_store[i];
  for(int c=0;c<4;c++){obs_store[i][c]=73;is_store[i][c]=scenario==3?(i%2):0;}
 }
}
static unsigned long long bench(int newer,int scenario,int count,unsigned long long*sum){
 unsigned long long a=ns(),acc=0;int current=73;
 for(int i=0;i<count;i++){
  unsigned long long value=0;int r;unsigned long long start=scenario==1?3000:10,end=2600;
  if(scenario<2){if(newer)r=new_global(1,123,start,&value,end,prev_cb_ts,lastest_ts,obvs,&last_cb_ts,&current);else r=old_global(1,123,start,&value,end,prev_cb_ts,lastest_ts,obvs,&last_cb_ts,&current);}
  else{if(newer)r=new_cluster(1,123,start,&value,end,0);else r=old_cluster(1,123,start,&value,end,0);}
  acc+=value+(unsigned)r;
 }
 unsigned long long elapsed=ns()-a;sink=acc;*sum=acc;return elapsed;
}
int main(void){
 puts("scenario,round,order,variant,iterations,elapsed_ns,checksum");
 for(int scenario=0;scenario<4;scenario++){
  setup(scenario);unsigned long long a,b;
  bench(0,scenario,5000,&a);bench(1,scenario,5000,&b);assert(a==b);
  int n=2000;while(n<200000){unsigned long long t=bench(0,scenario,n,&a);if(t>=20000000ULL)break;n*=2;}
  for(int round=0;round<31;round++){
   unsigned long long sums[2];for(int order=0;order<2;order++){
    int newer=(round+order)%2;unsigned long long t=bench(newer,scenario,n,&sums[newer]);
    printf("%d,%d,%d,%s,%d,%llu,%llu\n",scenario,round,order,newer?"candidate":"parent",n,t,sums[newer]);
   }assert(sums[0]==sums[1]);
  }
 }
 return 0;
}
'''
(L/'benchmark.c').write_text(s)
cmd=['gcc','-std=gnu11','-O2','-g','-ftrivial-auto-var-init=zero','-fno-builtin-calloc','-fno-builtin-free',str(L/'benchmark.c'),'-o',str(L/'benchmark')]
metadata={'date_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'scope':'x86_64 homelab VM, userspace calloc/free; no kernel allocator, real spinlocks, tracing or CPU-frequency updater; not phone performance','baseline_source_commit':'313c21a9dc2358667e9640e35738d2e9cab49a5e','candidate_source_commit':'800e6f1cf0c85c41bb60baa56a3cb17d4898214d','function_source_fixture_sha256':hashlib.sha256(sourcefile.read_bytes()).hexdigest(),'benchmark_source_sha256':hashlib.sha256(s.encode()).hexdigest(),'platform':platform.uname()._asdict(),'compiler':subprocess.check_output(['gcc','--version'],text=True).splitlines()[0],'command':cmd,'loadavg_before':list(os.getloadavg()),'cpu_model':next((x.split(':',1)[1].strip() for x in Path('/proc/cpuinfo').read_text().splitlines() if x.startswith('model name')),None),'sanitizers':False,'cpu_affinity':None,'warmup_calls_each_variant':5000,'paired_rounds':31,'phone_measured':False}
with (L/'compile.log').open('w') as f:subprocess.run(cmd,stdout=f,stderr=subprocess.STDOUT,check=True)
# Pin only this benchmark process, without changing governors/services.
allowed=sorted(os.sched_getaffinity(0));cpu=allowed[-1];metadata['cpu_affinity']=[cpu]
(L/'metadata.json').write_text(json.dumps(metadata,indent=2)+'\n')
with (L/'samples.csv').open('w') as f,(L/'run.stderr').open('w') as e:r=subprocess.run(['taskset','-c',str(cpu),str(L/'benchmark')],stdout=f,stderr=e)
metadata['exit_code']=r.returncode;metadata['loadavg_after']=list(os.getloadavg());(L/'metadata.json').write_text(json.dumps(metadata,indent=2)+'\n');assert r.returncode==0
import csv
rows=list(csv.DictReader((L/'samples.csv').open()));report=[]
labels=['global_full_history','global_early_exit','cluster_full_history','cluster_half_isolated']
for scenario,label in enumerate(labels):
 groups={tag:{int(x['round']):int(x['elapsed_ns'])/int(x['iterations']) for x in rows if int(x['scenario'])==scenario and x['variant']==tag} for tag in ['parent','candidate']}
 assert all(len(g)==31 for g in groups.values());changes=[100*(groups['parent'][i]-groups['candidate'][i])/groups['parent'][i] for i in range(31)]
 item={'scenario':label,'ns_per_call':{},'paired_time_reduction_percent_median':statistics.median(changes),'paired_time_reduction_percent_range':[min(changes),max(changes)]}
 for tag,v in groups.items():
  vals=sorted(v.values());q=statistics.quantiles(vals,n=4,method='inclusive');item['ns_per_call'][tag]={'median':statistics.median(vals),'q1':q[0],'q3':q[2],'min':vals[0],'max':vals[-1]}
 report.append(item)
(L/'summary.json').write_text(json.dumps({'scope':metadata['scope'],'results':report,'phone_measured':False},indent=2)+'\n')
(L/'SHA256SUMS').write_text(''.join(hashlib.sha256(f.read_bytes()).hexdigest()+'  '+f.name+'\n' for f in sorted(L.iterdir()) if f.is_file() and f.name!='SHA256SUMS'))
print(json.dumps(report,indent=2))
